ProjectDisbursementService.cs 
public class DisbursementService : IDisbursementService
{
    private readonly IApplicationDbContext _context;
    private readonly IKafkaProducer _kafkaProducer;
    private readonly IHubContext<NotificationHub> _hubContext;

    public async Task<Result> ApproveDisbursementAsync(
        Guid projectId, DisbursementDto dto)
    {
        // 1. Khởi tạo giao dịch & cập nhật kế hoạch vốn
        await using var tx = await _context.Database.BeginTransactionAsync();
        
        // 2. Phát sinh sự kiện bất đồng bộ qua Kafka broker
        await _kafkaProducer.PublishAsync(
            "erp.disbursement.approved", 
            new { ProjectId = projectId, Amount = dto.Amount });

        // 3. Đẩy thông báo tức thời qua WebSocket SignalR
        await _hubContext.Clients.Group(projectId.ToString())
            .SendAsync("ReceiveProgressUpdate", dto);

        await tx.CommitAsync();
        return Result.Success();
    }
}

Enterprise ERP — Phân Hệ Quản Lý Dự Án Xây Dựng & Nhân Sự (HR)

Hệ thống ERP giải quyết trọn vẹn vòng đời dự án thi công xây dựng: từ kế hoạch vốn, giải ngân, quản lý hồ sơ thầu, nghiệm thu sản lượng hợp đồng đến phân quyền nhân sự đa tầng.

Trách nhiệm & Nghiệp vụ đã xây dựng:

  • Quản lý dự án thi công: Xây dựng API hồ sơ dự án, trạng thái thi công, báo cáo nhật trình, vấn đề phát sinh tại công trường và bộ lọc đa tiêu chí theo chuyên đề chuyên sâu.
  • Theo dõi giải ngân & Kế hoạch vốn: Xây dựng luồng API theo dõi kế hoạch phân bổ vốn, thống kê giải ngân theo từng giai đoạn và xuất báo cáo quản trị cấp cao.
  • Luồng hồ sơ & Văn bản dự án: Xử lý văn bản đi/đến, hồ sơ pháp lý công trình, tài liệu mời thầu, hợp đồng và bảo lãnh ngân hàng.
  • Sản lượng & Tiến độ lũy kế: Tính toán sản lượng công việc theo hợp đồng, hỗ trợ theo dõi tiến độ thi công thực tế so với cam kết hợp đồng.
  • Quản trị nhân sự & Phân quyền RBAC: Quản lý cơ cấu phòng ban, danh sách nhân sự, phân vai trò (Roles) và ma trận quyền hạn (Permissions) chặt chẽ.
  • Real-time & Message Queue: Tích hợp Apache Kafka để đồng bộ sự kiện giữa các module và SignalR cho thông báo đẩy/thảo luận trực tiếp trong từng dự án.
